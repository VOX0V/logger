
URL du repo GitHub : https://github.com/VOX0V/logger

Docker compose :

services:
  logger:
    image: ghcr.io/vox0v/logger:latest
    container_name: Logger_8000
    restart: unless-stopped
    environment:
      - TZ=America/Toronto
    user: "1000:1000"
    ports:
      - "8000:8000"
    env_file:
      - .env
    volumes:
      # Base SQLite + données persistantes
      - /home/srv/gitapps/test/logger/appdata:/app/appdata
      - /home/srv/gitapps/test/logger/users:/app/users

GitHub push :

cd /home/srv/gitapps/logger_app
git add .
git commit -m "update"
git push origin main

Docker recreate :

cd /home/srv/gitapps/test
docker compose pull
docker compose up -d --force-recreate
